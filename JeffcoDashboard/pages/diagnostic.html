Layout Diagnostic Tool

Open your browser console and run this code to check the layout:

// Check #container
const container = document.getElementById('container');
console.log('=== #container ===');
console.log('display:', window.getComputedStyle(container).display);
console.log('flex-direction:', window.getComputedStyle(container).flexDirection);
console.log('flex:', window.getComputedStyle(container).flex);
console.log('position:', window.getComputedStyle(container).position);
console.log('width:', window.getComputedStyle(container).width);
console.log('offsetWidth:', container.offsetWidth);

// Check #map-container
const mapContainer = document.getElementById('map-container');
console.log('\n=== #map-container ===');
console.log('display:', window.getComputedStyle(mapContainer).display);
console.log('flex:', window.getComputedStyle(mapContainer).flex);
console.log('position:', window.getComputedStyle(mapContainer).position);
console.log('width:', window.getComputedStyle(mapContainer).width);
console.log('offsetWidth:', mapContainer.offsetWidth);
console.log('inline styles:', mapContainer.style.cssText);

// Check #map-sidebar
const mapSidebar = document.getElementById('map-sidebar');
console.log('\n=== #map-sidebar ===');
console.log('display:', window.getComputedStyle(mapSidebar).display);
console.log('flex:', window.getComputedStyle(mapSidebar).flex);
console.log('position:', window.getComputedStyle(mapSidebar).position);
console.log('z-index:', window.getComputedStyle(mapSidebar).zIndex);
console.log('width:', window.getComputedStyle(mapSidebar).width);
console.log('offsetWidth:', mapSidebar.offsetWidth);
console.log('offsetLeft:', mapSidebar.offsetLeft);
console.log('inline styles:', mapSidebar.style.cssText);

// Check parent
console.log('\n=== Parent of #map-sidebar ===');
console.log('parent:', mapSidebar.parentElement.id);
console.log('parent display:', window.getComputedStyle(mapSidebar.parentElement).display);
console.log('parent flex-direction:', window.getComputedStyle(mapSidebar.parentElement).flexDirection);

// Check siblings
console.log('\n=== Siblings of #map-sidebar ===');
const siblings = Array.from(mapSidebar.parentElement.children);
siblings.forEach((sibling, i) => {
  console.log(`Sibling ${i}:`, sibling.id || sibling.className, 
    'display:', window.getComputedStyle(sibling).display,
    'flex:', window.getComputedStyle(sibling).flex,
    'width:', window.getComputedStyle(sibling).width);
});
Instructions:
  1. Open your browser's Developer Tools (F12)
  2. Go to the Console tab
  3. Copy and paste the code above into the console
  4. Press Enter
  5. Share the output with me
Also check:
  1. Right-click on #map-sidebar → Inspect Element
  2. In the Elements panel, look at the computed styles
  3. Check if there's a red strikethrough on any flex-related properties (indicating they're being overridden)
  4. Check the "Layout" tab (if available) to see the flex container structure
